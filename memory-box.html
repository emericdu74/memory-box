<script>

const SCRIPT_URL = 'https://script.google.com/macros/s/AKfycbwyrFlQYwSjt0HpGMyKJfPfWGiuwLco0jZMFXjCe_AlWLjGZZuq0mQsg9XeTU-VHcbz/exec';

const SEQUENCE = ['Demain','Impair','Lundi','Premier_du_mois','3_mois','6_mois','Un_an'];

const SENS_NORMAL = {Demain:true,Impair:false,Lundi:true,Premier_du_mois:false,'3_mois':true,'6_mois':false,Un_an:true};


let state = {database:[],feuilles:{},history:[]};

SEQUENCE.forEach(f => state.feuilles[f] = []);

let session = {feuille:null,cards:[],index:0,total:0,ok:0,rate:0,results:[],sensNormal:true};

let dbSelected = new Set();

let dbFiltered = [];


function loadLocal() {

  try {

    const s = localStorage.getItem('memorybox_v2');

    if (s) {

      const p = JSON.parse(s);

      if (p.database) state.database = p.database;

      if (p.feuilles) state.feuilles = p.feuilles;

      if (p.history) state.history = p.history;

      SEQUENCE.forEach(f => { if (!state.feuilles[f]) state.feuilles[f] = []; });

    }

  } catch(e) {}

}


function saveLocal() {

  try { localStorage.setItem('memorybox_v2', JSON.stringify(state)); } catch(e) {}

}


async function syncNow() {

  showLoader(true);

  document.getElementById('sync-status-text').textContent = 'Synchronisation...';

  try {

    const res = await fetch(SCRIPT_URL + '?action=read');

    const data = await res.json();

    if (data.feuilles) state.feuilles = data.feuilles;

    if (data.database) state.database = data.database;

    if (data.history) state.history = data.history;

    SEQUENCE.forEach(f => { if (!state.feuilles[f]) state.feuilles[f] = []; });

    saveLocal();

    document.getElementById('sync-status-text').textContent = 'OK ' + new Date().toLocaleTimeString('fr-FR',{hour:'2-digit',minute:'2-digit'});

    renderHome();

    showToast('Synchronisé ✓');

  } catch(e) {

    document.getElementById('sync-status-text').textContent = 'Erreur';

    showToast('Erreur de synchronisation');

  }

  showLoader(false);

}


async function pushToSheets() {

  try {

    await fetch(SCRIPT_URL, {method:'POST',body:JSON.stringify({action:'write',feuilles:state.feuilles,database:state.database})});

  } catch(e) {}

}


async function pushHistory(entry) {

  try {

    await fetch(SCRIPT_URL, {method:'POST',body:JSON.stringify({action:'saveHistory',entry})});

  } catch(e) {}

}


function showScreen(id) {

  document.querySelectorAll('.screen').forEach(s => s.classList.remove('active'));

  document.getElementById('screen-' + id).classList.add('active');

}


function showTab(tab) {

  if (tab === 'home') { renderHome(); showScreen('home'); }

  else if (tab === 'dash') { renderDashboard(); showScreen('dash'); }

}


function backToHome() { renderHome(); showScreen('home'); }

function showLoader(v) { document.getElementById('loader').classList.toggle('hidden', !v); }


function renderHome() {

  const dbCount = state.database.length;

  document.getElementById('db-card-container').innerHTML = `

    <div class="feuille-card database-card" onclick="openDatabase()">

      <div class="feuille-info"><h3>Data Base</h3><span>Sélectionner les mots à apprendre</span></div>

      <div class="feuille-badge db">${dbCount}</div>

    </div>`;

  const container = document.getElementById('feuilles-container');

  container.innerHTML = '';

  SEQUENCE.forEach(f => {

    const cards = state.feuilles[f] || [];

    const count = cards.length;

    const isEmpty = count === 0;

    const div = document.createElement('div');

    div.className = 'feuille-card' + (isEmpty ? ' empty' : '');

    div.innerHTML = `

      <div class="feuille-info"><h3>${f.replace(/_/g,' ')}</h3><span>${isEmpty ? 'Aucune carte' : count + ' carte' + (count > 1 ? 's' : '')}</span></div>

      <div class="feuille-badge ${isEmpty ? 'zero' : ''}">${count}</div>`;

    if (!isEmpty) div.onclick = () => startSession(f);

    container.appendChild(div);

  });

}


function openDatabase() {

  dbFiltered = [...state.database];

function renderDBList() {

  document.getElementById('db-counter').textContent = state.database.length + ' mots';

  document.getElementById('db-selected-count').textContent = dbSelected.size + ' sélectionné(s)';

  document.getElementById('btn-send-demain').textContent = 'Envoyer en Demain (' + dbSelected.size + ')';

  document.getElementById('btn-send-demain').disabled = dbSelected.size === 0;

  document.getElementById('db-list').innerHTML = dbFiltered.map(card => {

    const realIdx = state.database.indexOf(card);

    const sel = dbSelected.has(realIdx);

    return `<div class="db-item ${sel ? 'selected' : ''}" onclick="toggleSelect(${realIdx})">

      <div class="db-checkbox"></div>

      <div class="db-word"><h4>${card.face1}</h4><span>${card.face2}</span></div>

    </div>`;

  }).join('');

}


function toggleSelect(idx) {

  if (dbSelected.has(idx)) dbSelected.delete(idx); else dbSelected.add(idx);

  renderDBList();

}


function toggleSelectAll() {

  dbSelected = dbSelected.size === state.database.length ? new Set() : new Set(state.database.map((_,i) => i));

  renderDBList();

}


function filterDB() {

  const q = document.getElementById('db-search-input').value.toLowerCase();

  dbFiltered = state.database.filter(c => c.face1.toLowerCase().includes(q) || c.face2.toLowerCase().includes(q));

  renderDBList();

}


function sendToDemain() {

  if (dbSelected.size === 0) return;

  const toSend = [...dbSelected].map(i => state.database[i]);

  state.feuilles['Demain'] = [...(state.feuilles['Demain'] || []), ...toSend];

  saveLocal();

  pushToSheets();

  showToast(toSend.length + ' mot(s) envoyé(s) en Demain ✓');

  dbSelected = new Set();

  backToHome();

}


function shuffle(arr) {

  const a = [...arr];

  for (let i = a.length - 1; i > 0; i--) {

    const j = Math.floor(Math.random() * (i + 1));

    [a[i], a[j]] = [a[j], a[i]];

  }

  return a;

}


function startSession(feuille) {

  const cards = state.feuilles[feuille];

  if (!cards || cards.length === 0) return;

  session = {feuille,cards:shuffle([...cards]),index:0,total:cards.length,ok:0,rate:0,results:[],sensNormal:SENS_NORMAL[feuille] !== false};

  state.feuilles[feuille] = [];

  showScreen('flash');

  showCard();

}


function showCard() {

  const card = session.cards[session.index];

  document.getElementById('flash-feuille-name').textContent = session.feuille.replace(/_/g,' ');

  document.getElementById('flash-counter').textContent = (session.index + 1) + '/' + session.total;

  document.getElementById('card-label-q').textContent = session.sensNormal ? 'Mot' : 'Traduction';

  document.getElementById('card-question').textContent = session.sensNormal ? card.face1 : card.face2;

  document.getElementById('card-label-a').textContent = session.sensNormal ? 'Traduction' : 'Mot';

  document.getElementById('card-answer').textContent = session.sensNormal ? card.face2 : card.face1;

  document.getElementById('card-answer').classList.remove('visible');

  document.getElementById('btn-reveal').classList.remove('hidden');

  document.getElementById('btn-result-row').classList.add('hidden');

  document.getElementById('score-ok-mini').textContent = session.ok;

  document.getElementById('score-ko-mini').textContent = session.rate;

  document.getElementById('progress-fill').style.width = (session.index / session.total * 100) + '%';

}


function revealCard() {

  document.getElementById('card-answer').classList.add('visible');

  document.getElementById('btn-reveal').classList.add('hidden');

  document.getElementById('btn-result-row').classList.remove('hidden');

}


function answer(reussi) {

  const card = session.cards[session.index];

  session.results.push({card,reussi});

  if (reussi) session.ok++; else session.rate++;

  session.index++;

  if (session.index >= session.total) endSession(); else showCard();

}


function endSession() {

  session.results.forEach(({card,reussi}) => {

    const dest = reussi ? nextFeuille(session.feuille) : prevFeuille(session.feuille);

    if (!state.feuilles[dest]) state.feuilles[dest] = [];

    state.feuilles[dest].push(card);

  });

  const total = session.ok + session.rate;

  const pct = total > 0 ? Math.round((session.ok / total) * 100) : 0;

  const entry = {date:new Date().toLocaleString('fr-FR'),feuille:session.feuille,ok:session.ok,rate:session.rate,pct};

  state.history.unshift(entry);

  saveLocal();

  pushToSheets();

  pushHistory(entry);

  document.getElementById('result-feuille').textContent = session.feuille.replace(/_/g,' ');

  document.getElementById('res-ok').textContent = session.ok;

  document.getElementById('res-ko').textContent = session.rate;

  document.getElementById('res-pct').textContent = pct + '%';

  document.getElementById('result-emoji').textContent = pct >= 80 ? '🎉' : pct >= 50 ? '💪' : '📚';

  document.getElementById('progress-fill').style.width = '100%';

  showScreen('results');

}


function nextFeuille(c) { const i = SEQUENCE.indexOf(c); return SEQUENCE[Math.min(i+1,SEQUENCE.length-1)]; }

function prevFeuille(c) { const i = SEQUENCE.indexOf(c); return SEQUENCE[Math.max(i-1,0)]; }
